Script di Monitoraggio Risorse Linux

Codice sorgente Bash corretto e ottimizzato per il controllo ciclico di RAM, CPU e spazio su Disco.

#!/bin/bash

while true; do

    # Estrazione dati sull'utilizzo delle risorse
    DISCO_USATO=$(df /dev/sda1 | grep sda1 | awk '{print $5}' | tr -d '%')
    RAM_USATA=$(free -m | awk '/Mem:/ {print $3}')
    CPU_USATA=$(mpstat 1 1 | grep "Average" | awk '{print 100 - $NF}')

    # Controllo RAM (Soglia: 400 MB)
    if [ "$RAM_USATA" -gt 400 ]; then
        echo "ATTENZIONE: Utilizzo RAM elevato! Valore attuale: ${RAM_USATA}MB"
    else
        echo "RAM OK!!!!!!"
    fi

    # Controllo CPU (Soglia: 90%)
    if [ "$CPU_USATA" -gt 90 ]; then
        echo "ATTENZIONE: Utilizzo CPU critico! Valore attuale: ${CPU_USATA}%"
    else
        echo "CPU OK!!!!"
    fi

    # Controllo Disco (Soglia: 80%)
    if [ "$DISCO_USATO" -gt 80 ]; then
        echo "ATTENZIONE: Spazio disco in esaurimento! Valore attuale: ${DISCO_USATO}%"
    else
        echo "DISCO OK!!!!"
    fi

    # Attesa di 10 secondi prima del prossimo ciclo
    sleep 10
done
